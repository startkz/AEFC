\section{METHODOLOGY}

\subsection{System Model and Recovery Objective}
We consider a CPS partitioned into $N$ interacting subsystems. Subsystem $i$ has state $x_i\in\mathbb R^{n_i}$ and control input $u_i\in\mathbb R^{m_i}$. Its attacked dynamics are
\begin{equation}
\dot x_i=f_i(x_i,x_{-i})+g_i(x_i)(u_i+a_i^u)+w_i,
\label{eq:dynamics}
\end{equation}
and the received observation is
\begin{equation}
y_i=\mathcal H_i(x_i)+a_i^y,
\label{eq:obs}
\end{equation}
where $\mathcal H_i$ is the observation map, $a_i^y$ models observation/communication corruption, $a_i^u$ models corruption between the shield output and the plant input, and $w_i$ represents bounded modeling disturbance.

Let
\begin{equation}
\mathcal S_i=\{x_i:h_{i,\ell}(x_i)\ge0,\ \ell=1,\ldots,M_i\}
\end{equation}
denote the physical safe set and
\begin{equation}
\mathcal R_i=\{x_i:V_i(x_i)\le\varepsilon_i\}
\end{equation}
a recovery set defined by a Lyapunov candidate $V_i$. Each subsystem maintains a state uncertainty set $\widehat{\mathcal X}_i(t)$ and admissible attack/disturbance sets $\mathcal A_i^u(t)$ and $\mathcal W_i(t)$. The physical guarantees below are conditional on
\begin{equation}
x_i(t)\in\widehat{\mathcal X}_i(t),\quad
a_i^u(t)\in\mathcal A_i^u(t),\quad
w_i(t)\in\mathcal W_i(t).
\label{eq:coverage}
\end{equation}

\subsection{Threat Model and Trust Boundary}
The adversary may act at two layers.

\textbf{Physical-channel adversary.}
The attacker can bias, replay, delay, or partially suppress observations and can perturb the command after it leaves the controller, subject to the modeled attack envelope. An attacker that can impose an unbounded arbitrary actuator command is outside the guarantee.

\textbf{Federated recovery-knowledge adversary.}
At most $f$ of $N$ participating clients are Byzantine. Byzantine clients may collude and submit arbitrary update values, sparse supports, timestamps, prototypes, advantages, confidence values, or other self-reported metadata. They may also replay stale but syntactically valid recovery knowledge. The server follows the coordination protocol.

Authenticated provenance fields and client self-reported semantic fields are conceptually distinct. When cryptographic source authentication, secure time, or remote attestation is unavailable, the corresponding provenance property must be treated as an implementation assumption rather than inferred from a client-provided identifier.

The central security requirement is:
\begin{quote}
Corrupted federated knowledge may affect recovery utility, but its influence on belief, authorization, deployed-policy adaptation, and physical actuation must remain bounded by explicit mechanisms and assumptions.
\end{quote}

\subsection{Heterogeneous Local Recovery Learner}
Client $i$ uses
\begin{equation}
\pi_i(o_i)=g_{\xi_i}\!\left(b_{\theta}(e_{\psi_i}(o_i))\right),
\end{equation}
where $e_{\psi_i}$ is a local observation adapter, $b_{\theta}$ is the shared recovery backbone, and $g_{\xi_i}$ is a local action adapter. Only $\theta$ is federated. This design prevents direct averaging of incompatible subsystem input/output layers.

Let $Q_i(s,a)$ be the local recovery critic. The recovery advantage
\begin{equation}
A_i(s,a)=Q_i(s,a)-Q_i(s,\pi(s))
\end{equation}
is used only as a utility signal and never as a security credential.

\subsection{Provenance-Aware Temporal Belief Update}
Let evidence from source $j$ at time $t$ be
\begin{equation}
e_{j,t}=(z_{j,t},p_{j,t}),
\end{equation}
where $z_{j,t}$ is the recovery evidence and
\begin{equation}
p_{j,t}=(\mathrm{src}_j,\tau_{j,t},v_{j,t}^{\mathrm{model}},\ell_{j,t})
\end{equation}
contains provenance attributes such as source, timestamp, model version, and lineage.

Let $\chi_{j,t}\in\{0,1\}$ denote provenance validity under the deployed authenticity mechanism, $\Delta_{j,t}=t-\tau_{j,t}$ the evidence age, and
\begin{equation}
r_{j,t}=d\!\left(z_{j,t},\mathcal C(b_{t-1},\widehat{\mathcal X}_t)\right)
\end{equation}
a physics/temporal consistency residual. We define a bounded influence weight
\begin{equation}
\omega_{j,t}=\chi_{j,t}e^{-\beta\Delta_{j,t}}\psi_c(r_{j,t}),
\qquad 0\le\psi_c(\cdot)\le1.
\end{equation}
A robust recovery-knowledge summary is
\begin{equation}
\bar k_t=\mathcal R_f(\{\omega_{j,t}z_{j,t}\}_{j=1}^{N}),
\end{equation}
and the recovery belief is updated by
\begin{equation}
b_{t+1}=\Pi_{\mathcal B}T(b_t,\widehat y_t,\bar k_t).
\label{eq:belief}
\end{equation}
We take $\mathcal B$ to be closed and convex and $\Pi_{\mathcal B}$ to be the Euclidean projection; hence the projection used in the analysis is non-expansive.

\begin{assumption}[Contractive recovery-belief dynamics]
There exist $0\le\lambda_b<1$, $L_y\ge0$, and $L_k\ge0$ such that
\begin{align}
\|T(b,y,k)-T(b',y',k')\|
\le&\ \lambda_b\|b-b'\| \nonumber\\
&+L_y\|y-y'\|+L_k\|k-k'\|.
\end{align}
\end{assumption}

\begin{theorem}[Bounded belief distortion]
Let $b_t^\circ$ denote the clean counterfactual belief. If
$\|a_t^y\|\le\epsilon_y$ and
$\|\bar k_t-\bar k_t^\circ\|\le\epsilon_k$ for all $t$, then
\begin{align}
\|b_t-b_t^\circ\|
\le&\ \lambda_b^t\|b_0-b_0^\circ\| \nonumber\\
&+\frac{L_y\epsilon_y+L_k\epsilon_k}{1-\lambda_b}
(1-\lambda_b^t).
\label{eq:belief-bound}
\end{align}
\end{theorem}
\begin{proof}
Projection onto $\mathcal B$ is non-expansive. Applying the Lipschitz condition to the attacked and clean recursions gives
$e_{t+1}\le\lambda_b e_t+L_y\epsilon_y+L_k\epsilon_k$.
Unrolling the recursion yields \eqref{eq:belief-bound}.
\end{proof}

If poisoning stops at $T_p$ and the combined post-attack discrepancy entering the belief recursion is bounded by $\epsilon_{\mathrm{clean}}$, the same recursion gives geometric correction toward an $\epsilon_{\mathrm{clean}}/(1-\lambda_b)$ residual neighborhood. Exact finite-time return is not claimed without zero residual disturbance or an additional reset/rollback mechanism.

\subsection{Certificate-Supported Local Recovery Knowledge}
For an honest client, a local transition $\tau=(s,a,r,s')$ is eligible for federation only when it is recovery-useful and consistent with local safety/recovery evidence. Define
\begin{align}
m_i^{\mathrm{safe}}(\tau)
=&\ \min_\ell\inf_{\substack{x\in\widehat{\mathcal X}_i\\a^u\in\mathcal A_i^u\\w\in\mathcal W_i}}
\Big[\nabla h_{i,\ell}^{\top}(f_i+g_i(u+a^u)+w) \nonumber\\
&\hspace{5em}+\alpha_\ell(h_{i,\ell})\Big]
\end{align}
and
\begin{equation}
m_i^{\mathrm{rec}}(\tau)=-
\sup_{\substack{x\in\widehat{\mathcal X}_i\\a^u\in\mathcal A_i^u\\w\in\mathcal W_i}}
[\dot V_i(x,u,a^u,w)+c_iV_i(x)].
\end{equation}
The honest-client sharing set is
\begin{equation}
\mathcal B_i^+=
\{\tau:A_i(\tau)>\zeta_A,\ 
m_i^{\mathrm{safe}}(\tau)\ge0,\ 
m_i^{\mathrm{rec}}(\tau)>\zeta_V\}.
\end{equation}
This gate limits what an honest client shares. It does not authenticate Byzantine certificates.

\subsection{Representation Alignment and Sparse Coordination}
For coarse recovery mode $m$, honest client $i$ may construct a latent prototype
\begin{equation}
p_{i,m}=\frac{1}{|\mathcal B_{i,m}|}\sum_{o\in\mathcal B_{i,m}}e_{\psi_i}(o)
\end{equation}
and use a robust center $\bar p_m$ only to reduce honest-client representation drift.

For communication, let $C_k(v)$ retain the $k$ largest-magnitude coordinates. With error feedback,
\begin{align}
z_i^t&=g_i^t+e_i^t,\\
U_i^t&=C_k(z_i^t),\\
e_i^{t+1}&=z_i^t-U_i^t.
\end{align}

\begin{lemma}[Deterministic Top-$k$ contraction]
For any $v\in\mathbb R^d$,
\begin{equation}
\|v-C_k(v)\|_2^2
\le\left(1-\frac{k}{d}\right)\|v\|_2^2.
\end{equation}
\end{lemma}

The contraction result does not justify treating an untransmitted coordinate as an observed numerical zero in a Byzantine proof. To keep the aggregation semantics well-defined, the preferred implementation uses a common support $S_t$, $|S_t|=k$, determined independently of any single client's report; all participating clients then upload $g_{i,t}[S_t]$. If client-specific supports are used instead, the server must employ a support-aware robust estimator and the corresponding theorem must be changed accordingly.

On a common coordinate population, the server applies a Byzantine-resilient estimator $\mathcal R_f$. For coordinate-wise $f$-trimmed mean, if all honest values on a coordinate lie in
$[\mu^H_{t,j}-r_{t,j},\mu^H_{t,j}+r_{t,j}]$ and $f<N/2$, then
\begin{equation}
|\widehat g_{t,j}-\mu^H_{t,j}|\le r_{t,j}.
\end{equation}
This envelope is the source of $\epsilon_k$ used in the belief analysis; client-reported confidence or advantage is not.

\subsection{Risk-Aware Recovery Gate}
The belief and learned policy produce a candidate $u_t^\pi$. Before this proposal is allowed to exercise recovery authority, AEFC computes a robust safety shortfall
\begin{equation}
R_t^{\mathrm{safe}}(u)=
\max_\ell
\sup_{\substack{x\in\mathcal X(b_t)\\
a^u\in\mathcal A_t^u\\
w\in\mathcal W_t}}
[-\mathcal B_\ell(x,u,a^u,w)]_+,
\end{equation}
where
\begin{align}
\mathcal B_\ell(x,u,a^u,w)
=&\ \nabla h_\ell(x)^\top
[f(x)+g(x)(u+a^u)+w] \nonumber\\
&+\alpha_\ell(h_\ell(x)).
\end{align}
We combine this term with recovery and uncertainty penalties:
\begin{align}
\widehat{\mathfrak R}_t(u)=&
\lambda_sR_t^{\mathrm{safe}}(u)
+\lambda_r[-m_t^{\mathrm{rec}}(u)]_+ \nonumber\\
&+\lambda_b\,\mathrm{diam}(\mathcal X(b_t))
+\lambda_p(1-q_t),
\end{align}
where $q_t\in[0,1]$ is a protocol-level trust state derived from provenance validity, age, cross-source consistency, and belief uncertainty.

The learned proposal is authorized only when
\begin{equation}
G_t(u)=
\mathbf 1\{\widehat{\mathfrak R}_t(u)\le\rho_{\max}-\epsilon_R\}
\mathbf 1\{q_t\ge q_{\min}\}.
\end{equation}
Then
\begin{equation}
u_t^{\mathrm{prop}}=
\begin{cases}
u_t^\pi,&G_t=1,\\
u_t^{\mathrm{fallback}},&G_t=0.
\end{cases}
\end{equation}

\begin{theorem}[Bounded false authorization]
Assume
\begin{equation}
\Pr[
\mathfrak R_t^{\mathrm{true}}(u)
\le\widehat{\mathfrak R}_t(u)+\epsilon_R]\ge1-\delta_g.
\end{equation}
Then
\begin{equation}
\Pr[G_t(u)=1\ \wedge\
\mathfrak R_t^{\mathrm{true}}(u)>\rho_{\max}]
\le\delta_g.
\end{equation}
\end{theorem}
If all uncertainty sets are deterministic outer approximations, $\delta_g$ can be zero.

\subsection{Trust-Gated Online Policy Adaptation}
AEFC separates a candidate model from the deployed controller. Given robust aggregate $g_t$, the candidate is
\begin{equation}
\widetilde\theta_{t+1}=
\Pi_{\mathbb B(\theta_t^{\mathrm{dep}},\Delta_\theta)}
\left[\theta_t^{\mathrm{dep}}
-\eta_t q_t\,\operatorname{clip}_C(g_t)\right].
\end{equation}
Deployment requires
\begin{equation}
A_t=
\mathbf 1\{q_t\ge q_{\min}\}
\mathbf 1\{G_t=1\}
\mathbf 1\{\mathrm{validation}_t=1\},
\end{equation}
and
\begin{equation}
\theta_{t+1}^{\mathrm{dep}}=
A_t\widetilde\theta_{t+1}
+(1-A_t)\theta_t^{\mathrm{dep}}.
\end{equation}

\begin{theorem}[Bounded deployed-policy deviation]
For every accepted round,
\begin{equation}
\|\theta_{t+1}^{\mathrm{dep}}-\theta_t^{\mathrm{dep}}\|
\le
\min\{\eta_tC,\Delta_\theta\}.
\end{equation}
If $\pi_\theta$ is $L_\pi$-Lipschitz in its parameters, then
\begin{equation}
\|\pi_{\theta_{t+1}^{\mathrm{dep}}}(x)
-\pi_{\theta_t^{\mathrm{dep}}}(x)\|
\le
L_\pi\min\{\eta_tC,\Delta_\theta\}.
\end{equation}
\end{theorem}

Finite correction after poisoning additionally requires either finite detection and rollback to a certified checkpoint, or a clean post-attack contraction.
\begin{corollary}[Finite $\varepsilon$-correction after poisoning]
Suppose poisoning stops at $T_p$ and, for $t\ge T_p$,
\begin{equation}
\|\theta_{t+1}-\theta^c\|
\le\kappa\|\theta_t-\theta^c\|+\epsilon_c,
\qquad 0\le\kappa<1.
\end{equation}
Then, for every integer $s\ge0$,
\begin{align}
\|\theta_{T_p+s}-\theta^c\|
\le&\ \kappa^s\|\theta_{T_p}-\theta^c\| \nonumber\\
&+\frac{1-\kappa^s}{1-\kappa}\epsilon_c.
\end{align}
Consequently, for any $\varepsilon>\epsilon_c/(1-\kappa)$, the deployed parameters enter the $\varepsilon$-ball around $\theta^c$ after finitely many clean rounds. Exact return requires $\epsilon_c=0$ or a certified rollback/reset mechanism.
\end{corollary}
\begin{proof}
Unrolling the contraction recursion for $s$ clean rounds gives the stated geometric bound. Since $\kappa^s\to0$, every radius strictly larger than the residual floor $\epsilon_c/(1-\kappa)$ is reached in finite time.
\end{proof}

\subsection{Robust Physical Execution Shield}
For compactness, let $x$ and $u$ denote the stacked subsystem state and input and let $\mathcal S=\prod_i\mathcal S_i$. Assume $f$ and $g$ are locally Lipschitz, each barrier function $h_\ell$ is continuously differentiable, and $\alpha_\ell$ is an extended class-$\mathcal K$ function. The authorized or fallback proposal is still not directly applied. AEFC solves
\begin{equation}
u_t^\star=
\arg\min_{u,\delta\ge0}
\|u-u_t^{\mathrm{prop}}\|_2^2+\lambda_\delta\delta^2
\label{eq:shield-new}
\end{equation}
subject to the hard robust safety constraints
\begin{align}
\inf_{\substack{x\in\widehat{\mathcal X}_t\\
a^u\in\mathcal A_t^u\\
w\in\mathcal W_t}}
\Big[&\nabla h_\ell(x)^\top(f(x)+g(x)(u+a^u)+w) \nonumber\\
&+\alpha_\ell(h_\ell(x))\Big]\ge0
\label{eq:robust-cbf}
\end{align}
and a soft recovery CLF constraint that may be relaxed by $\delta$; the safety constraints remain hard. The actual plant receives
\begin{equation}
u_t^{\mathrm{exec}}=u_t^\star+a_t^u.
\end{equation}

\begin{theorem}[Compositional safety under joint corruption]
Suppose \eqref{eq:coverage} holds for all $t$, the hard constraints in \eqref{eq:robust-cbf} are feasible, and $x(0)\in\mathcal S$. Then
\begin{equation}
x(t)\in\mathcal S,\qquad \forall t\ge0,
\end{equation}
for any recovery belief, federated knowledge, deployed policy, or learned proposal generated under poisoning, provided the realized physical corruption remains inside the modeled envelope.
\end{theorem}
\begin{proof}
For every admissible state, downstream control corruption, and disturbance, the robust barrier constraint enforces
$\dot h_\ell+\alpha_\ell(h_\ell)\ge0$.
Standard forward-invariance arguments therefore preserve every
$\{h_\ell\ge0\}$ and hence their intersection. The proof is independent of the quality or origin of the learned proposal.
\end{proof}

\subsection{Guarantee Summary and Scope}
The security guarantees are intentionally separated:
\begin{align}
\text{G1: }&\text{bounded belief distortion},\\
\text{G2: }&\text{bounded false authorization},\\
\text{G3: }&\text{finite $\varepsilon$-correction above the residual floor} \nonumber\\
&\text{under explicit contraction/rollback},\\
\text{G4: }&\text{bounded deployed-policy deviation},\\
\text{G5: }&\text{physical forward invariance}.
\end{align}
G5 does not require convergence of the federated learner. G1--G4 explain why federated recovery knowledge has bounded authority rather than being made irrelevant by the final shield.
